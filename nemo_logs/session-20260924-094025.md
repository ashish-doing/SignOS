```python
"""Pydantic models for the intent bus schema (signos.intent.v1)."""
from __future__ import annotations
from enum import Enum
from typing import Literal, Optional
from pydantic import BaseModel, Field, field_validator
import uuid
from datetime import datetime, timezone


class IntentKind(str, Enum):
    GESTURE = "gesture"
    ISL = "isl"


class GestureIntent(str, Enum):
    POINT = "POINT"
    PINCH = "PINCH"
    DOUBLE_PINCH = "DOUBLE_PINCH"
    PINCH_HOLD_MOVE = "PINCH_HOLD_MOVE"
    PINCH_HOLD_STILL = "PINCH_HOLD_STILL"
    SCROLL_V = "SCROLL_V"
    SCROLL_H = "SCROLL_H"
    SWIPE_LEFT = "SWIPE_LEFT"
    SWIPE_RIGHT = "SWIPE_RIGHT"
    PALM_HOLD = "PALM_HOLD"
    CLUTCH_ON = "CLUTCH_ON"
    CLUTCH_OFF = "CLUTCH_OFF"
    NONE = "NONE"
    TRANSITION = "TRANSITION"


class HandLabel(str, Enum):
    LEFT = "left"
    RIGHT = "right"
    TWO = "two"


class Pointer(BaseModel):
    x: float = Field(..., ge=0.0, le=1.0)
    y: float = Field(..., ge=0.0, le=1.0)


class IntentRecord(BaseModel):
    schema: Literal["signos.intent.v1"] = "signos.intent.v1"
    ts: str  # ISO8601 UTC with ms
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    kind: IntentKind
    intent: str  # GestureIntent or "ISL_<NAME>"
    confidence: float = Field(..., ge=0.0, le=1.0)
    pointer: Optional[Pointer] = None
    hand: HandLabel
    raw_duration_ms: int = Field(..., ge=0)
    source_fps: float = Field(..., ge=0.0)

    @field_validator("ts", mode="before")
    @classmethod
    def _ensure_utc_iso(cls, v: str | datetime) -> str:
        if isinstance(v, datetime):
            if v.tzinfo is None:
                v = v.replace(tzinfo=timezone.utc)
            return v.isoformat(timespec="milliseconds").replace("+00:00", "Z")
        return v

    @field_validator("intent")
    @classmethod
    def _validate_intent(cls, v: str) -> str:
        # Allow GestureIntent values or ISL_* pattern
        try:
            GestureIntent(v)
            return v
        except ValueError:
            if v.startswith("ISL_"):
                return v
            raise ValueError(f"Invalid intent: {v}")

    def to_json_line(self) -> str:
        return self.model_dump_json(exclude_none=True)
```

```python
"""Intent bus writer — single JSON line per intent, flushed immediately."""
from __future__ import annotations
import json
import os
import threading
from pathlib import Path
from typing import Optional

from .schemas import IntentRecord


class IntentBus:
    """Thread-safe, append-only, line-delimited JSON writer with fsync."""

    def __init__(self, path: str | os.PathLike = "signos/state/intent_bus.jsonl"):
        self._path = Path(path)
        self._path.parent.mkdir(parents=True, exist_ok=True)
        self._lock = threading.Lock()
        # Open in append, unbuffered text mode (flush + os.fsync on each write)
        self._fp = open(self._path, "a", encoding="utf-8", buffering=1)

    def write(self, record: IntentRecord) -> None:
        line = record.to_json_line() + "\n"
        with self._lock:
            self._fp.write(line)
            self._fp.flush()
            os.fsync(self._fp.fileno())

    def close(self) -> None:
        with self._lock:
            if not self._fp.closed:
                self._fp.flush()
                os.fsync(self._fp.fileno())
                self._fp.close()

    def __enter__(self) -> "IntentBus":
        return self

    def __exit__(self, exc_type, exc_val, exc_tb) -> None:
        self.close()


# Convenience singleton for scripts
_default_bus: Optional[IntentBus] = None


def get_default_bus() -> IntentBus:
    global _default_bus
    if _default_bus is None:
        _default_bus = IntentBus()
    return _default_bus


def emit_idle_nonce(bus: IntentBus, fps: float = 0.0) -> None:
    """Emit a single NONE intent (idle frame)."""
    from datetime import datetime, timezone
    record = IntentRecord(
        ts=datetime.now(timezone.utc),
        kind=IntentRecord.__fields__["kind"].default,  # type: ignore
        intent="NONE",
        confidence=1.0,
        pointer=None,
        hand="right",
        raw_duration_ms=0,
        source_fps=fps,
    )
    # Fix kind enum default
    record.kind = "gesture"  # type: ignore
    bus.write(record)
```

```python
"""Camera capture with live FPS reporting."""
from __future__ import annotations
import cv2
import time
from typing import Generator, Tuple, Optional
import numpy as np


class CameraCapture:
    """OpenCV camera wrapper yielding (frame_bgr, fps) tuples."""

    def __init__(
        self,
        device_index: int = 0,
        width: int = 640,
        height: int = 480,
        fps_target: int = 30,
    ):
        self.device_index = device_index
        self.width = width
        self.height = height
        self.fps_target = fps_target
        self._cap: Optional[cv2.VideoCapture] = None
        self._prev_time = time.perf_counter()
        self._fps_smooth = 0.0

    def __enter__(self) -> "CameraCapture":
        self._cap = cv2.VideoCapture(self.device_index, cv2.CAP_DSHOW)
        if not self._cap.isOpened():
            raise RuntimeError(f"Cannot open camera index {self.device_index}")
        self._cap.set(cv2.CAP_PROP_FRAME_WIDTH, self.width)
        self._cap.set(cv2.CAP_PROP_FRAME_HEIGHT, self.height)
        self._cap.set(cv2.CAP_PROP_FPS, self.fps_target)
        # Warm-up: drop first few frames
        for _ in range(3):
            self._cap.read()
        self._prev_time = time.perf_counter()
        return self

    def __exit__(self, exc_type, exc_val, exc_tb) -> None:
        if self._cap is not None:
            self._cap.release()
            self._cap = None

    def frames(self) -> Generator[Tuple[np.ndarray, float], None, None]:
        """Yield (frame_bgr, current_fps)."""
        if self._cap is None:
            raise RuntimeError("Camera not opened. Use 'with CameraCapture() as cap:'")
        while True:
            ok, frame = self._cap.read()
            if not ok:
                raise RuntimeError("Camera read failed")
            now = time.perf_counter()
            dt = now - self._prev_time
            self._prev_time = now
            inst_fps = 1.0 / dt if dt > 0 else 0.0
            # EMA smoothing alpha=0.1
            self._fps_smooth = 0.9 * self._fps_smooth + 0.1 * inst_fps
            yield frame, self._fps_smooth
```

```python
"""Batch 1 demo: open camera, print FPS, emit idle NONE to intent bus."""
from __future__ import annotations
import argparse
import signal
import sys
import time

from signos.src.perception.capture import CameraCapture
from signos.src.perception.intent_bus import IntentBus, emit_idle_nonce


def main() -> int:
    parser = argparse.ArgumentParser(description="SignOS capture + idle intent bus")
    parser.add_argument("--camera", type=int, default=0, help="Camera device index")
    parser.add_argument("--width", type=int, default=640)
    parser.add_argument("--height", type=int, default=480)
    parser.add_argument("--fps", type=int, default=30)
    parser.add_argument("--bus", type=str, default="signos/state/intent_bus.jsonl")
    args = parser.parse_args()

    bus = IntentBus(args.bus)
    shutdown = False

    def _sigint(_sig, _frm):
        nonlocal shutdown
        shutdown = True

    signal.signal(signal.SIGINT, _sigint)
    signal.signal(signal.SIGTERM, _sigint)

    print(f"[SignOS] Opening camera {args.camera} @ {args.width}x{args.height}...")
    try:
        with CameraCapture(args.camera, args.width, args.height, args.fps) as cap:
            print("[SignOS] Running — press Ctrl+C to stop")
            frame_count = 0
            t0 = time.perf_counter()
            for frame, fps in cap.frames():
                if shutdown:
                    break
                emit_idle_nonce(bus, fps)
                frame_count += 1
                if frame_count % 30 == 0:
                    elapsed = time.perf_counter() - t0
                    print(f"  frames={frame_count}  fps={fps:.1f}  bus={args.bus}")
    except RuntimeError as e:
        print(f"[SignOS] ERROR: {e}", file=sys.stderr)
        return 1
    finally:
        bus.close()
        print("[SignOS] Stopped.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def test_bus_thread_safety():
    import threading
    with tempfile.TemporaryDirectory() as td:
        path = Path(td) / "bus.jsonl"
        bus = IntentBus(path)

        def writer(n: int):
            for i in range(n):
                rec = IntentRecord(
                    ts="2026-09-24T10:15:32.123Z",
                    kind=IntentKind.GESTURE,
                    intent=GestureIntent.NONE,
                    confidence=1.0,
                    pointer=None,
                    hand=HandLabel.RIGHT,
                    raw_duration_ms=0,
                    source_fps=30.0,

                bus.write(rec)

        lines = path.read_text(encoding="utf-8").strip().splitlines()
```
